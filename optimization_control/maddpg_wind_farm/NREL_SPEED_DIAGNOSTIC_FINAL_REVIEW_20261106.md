```python
from pathlib import Path
import json
import numpy as np
p = Path('./optimization_control/experiments/20261106_nrel_speed_diagnostic')
anchors = json.loads((p/'anchors.json').read_text())
complete = json.loads((p/'complete.json').read_text())
delta, label_delta = [], []
for a, item in zip(anchors, complete['outcomes'], strict=True):
    with np.load(a['source_trace']['path'], allow_pickle=False) as z:
        row = z['trace'][a['case_index'], a['t']]
    op = row[19:28].reshape(3, 3)
    point = np.r_[row[4:7], op[:, 0], op[:, 1], op[:, 2]].astype(np.float32)
    with np.load(item['artifact']['path'], allow_pickle=False) as z:
        x, y, label, idx = [z[k] for k in ('inputs', 'outputs', 'labels', 'mapping')]
    np.testing.assert_array_equal(x[idx[0]], point)
    np.testing.assert_array_equal(x[:, :9], np.broadcast_to(point[:9], (len(x), 9)))
    np.testing.assert_array_equal(y[idx[0]], np.r_[row[13:16], row[10:13]].astype(np.float32))
    farm = y[:, :3].astype(float).sum(axis=1)
    teacher = label[:, :3].astype(float).sum(axis=1)
    delta.append((farm[idx[2]]-farm[idx[1]])/1000)
    label_delta.append((teacher[idx[2]]-teacher[idx[1]])/1000)
print('anchors', len(delta), 'model_mean_kW', np.mean(delta), 'label_mean_kW', np.mean(label_delta))
print('model_up_down', sum(d > 0 for d in delta), sum(d < 0 for d in delta), 'label_up', sum(d > 0 for d in label_delta))
```
